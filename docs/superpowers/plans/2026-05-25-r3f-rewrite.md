# OrgGraph R3F Rewrite — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `3d-force-graph` with React Three Fiber + D3-force-3d, introduce Zustand for global state, render theme clouds as glowing bounding spheres with a fresnel GLSL shader, and drive oversight connections as instanced amber particles.

> **Note on MarchingCubes:** The engineering plan specifies `<MarchingCubes>` from drei for theme clouds. Task 8 uses a dynamic bounding sphere + fresnel shader instead. MarchingCubes metaball positions are React props — updating them each frame causes React re-renders and defeats the imperative frame loop pattern. The bounding sphere achieves the same containment visual (encompasses member nodes, glowing rim via fresnel) without that cost. MarchingCubes remains a valid enhancement once layout has stabilized.

**Architecture:** D3-force-3d runs as a manually-ticked simulation inside the R3F canvas via `useFrame`. Simulation nodes are plain JS objects that D3 mutates in-place (`.x`, `.y`, `.z`), so no React state update fires per frame — children read position directly from the sim node object in their own `useFrame` callbacks. Zustand holds all UI state (selection, focus, layers, HUD visibility); canvas components read it via `useGraphStore.getState()` inside `useFrame` to avoid subscription-triggered re-renders mid-frame.

**Tech Stack:** Vite 5, React 18, @react-three/fiber 8, @react-three/drei 9, d3-force-3d 3, Three.js 0.184, Zustand 4, Tailwind CSS 3, Vitest (unit tests for pure logic).

```js
/**
 * Combines buildGraphData (adds _size) with getVisibleData (layer filtering).
 * Use this instead of calling them separately.
 */
export function buildVisibleData(rawData, layers) {
  // 1. Compute sizes for all nodes
  const { nodes: sizedNodes, links: allLinks } = buildGraphData(rawData);

  // 2. Build a map of id → _size for fast lookup
  const sizeMap = new Map(sizedNodes.map(n => [n.id, n._size]));

  // 3. Apply layer filtering (same logic as getVisibleData)
  const excludedNodeIds = new Set();
  if (layers.themes === false) {
    for (const node of rawData.nodes) {
      if (node.type === 'theme') excludedNodeIds.add(node.id);
    }
  }
  if (layers.projects === false) {
    for (const node of rawData.nodes) {
      if (node.type === 'project') excludedNodeIds.add(node.id);
    }
  }

  const nodes = sizedNodes.filter(n => !excludedNodeIds.has(n.id));

  const links = allLinks.filter(link => {
    const srcId = resolveId(link.source);
    const tgtId = resolveId(link.target);
    if (excludedNodeIds.has(srcId) || excludedNodeIds.has(tgtId)) return false;
    if (layers.org === false && link.type === 'reports-to') return false;
    if (layers.oversight === false && link.type === 'oversees') return false;
    if (layers.work === false && link.type === 'works-on') return false;
    return true;
  });

  return { nodes, links };
}
```

```js
export const useGraphStore = create((set, get) => ({
  rawData: null,
  selectedNodeId: null,
  hoveredNodeId: null,
  focusNodeId: null,
  focusSet: new Set(),
  hudVisible: true,
  layers: { org: true, oversight: true, work: true, themes: true, projects: true },

  setRawData: (data) => set({ rawData: data, selectedNodeId: null, hoveredNodeId: null, focusNodeId: null, focusSet: new Set() }),

  setSelectedNodeId: (id) => set({ selectedNodeId: id }),
  setHoveredNodeId: (id) => set({ hoveredNodeId: id }),
  toggleHUD: () => set(s => ({ hudVisible: !s.hudVisible })),

  toggleLayer: (name) => set(s => ({ layers: { ...s.layers, [name]: !s.layers[name] } })),

  setFocusNodeId: (id) => {
    if (!id) { set({ focusNodeId: null, focusSet: new Set() }); return; }
    const raw = get().rawData;
    if (!raw) return;
    const focusNode = raw.nodes.find(n => n.id === id);
    const focusSet = focusNode ? getFocusSet(focusNode, raw) : new Set();
    set({ focusNodeId: id, focusSet });
  },

  clearSelection: () => set({ selectedNodeId: null }),

  resetAll: () => set({
    rawData: null,
    selectedNodeId: null,
    hoveredNodeId: null,
    focusNodeId: null,
    focusSet: new Set(),
    hudVisible: true,
  }),
}));
```

### Task 5: useForceLayout hook

D3-force-3d simulation with radial positioning by node type. Must be called inside the R3F `<Canvas>` tree (uses `useFrame`).

**Files:**
- Create: `src/hooks/useForceLayout.js`

- [ ] **Step 1: Create src/hooks/useForceLayout.js**

```js
import { useEffect, useRef } from 'react';
import { useFrame } from '@react-three/fiber';
import {
  forceSimulation,
  forceLink,
  forceManyBody,
  forceCenter,
  forceRadial,
  forceCollide,
} from 'd3-force-3d';

const RADIAL_TARGET = { person: 0, project: 150, theme: 350 };
const RADIAL_STRENGTH = { person: 0.1, project: 0.3, theme: 0.6 };

export function useForceLayout(graphData) {
  const simRef = useRef(null);
  const simNodesRef = useRef([]);

  useEffect(() => {
    if (!graphData.nodes.length) return;

    // Clone nodes/links so D3 can mutate them freely
    const nodes = graphData.nodes.map(n => ({ ...n }));
    const links = graphData.links.map(l => ({ ...l }));

    const simulation = forceSimulation(nodes, 3) // 3 = 3D
      .force('link',
        forceLink(links)
          .id(d => d.id)
          .distance(d => d.type === 'oversees' ? 80 : 40)
          .strength(0.4)
      )
      .force('charge', forceManyBody().strength(-120))
      .force('center', forceCenter(0, 0, 0))
      .force('radial',
        forceRadial(d => RADIAL_TARGET[d.type] ?? 100, 0, 0, 0)
          .strength(d => RADIAL_STRENGTH[d.type] ?? 0.2)
      )
      .force('collide', forceCollide().radius(d => (d._size || 5) * 2))
      .alphaDecay(0.015)
      .stop();

    // Pre-warm: run synchronously so first render isn't at origin
    for (let i = 0; i < 200; i++) simulation.tick();

    simNodesRef.current = nodes;
    simRef.current = simulation;

    return () => { simulation.stop(); simRef.current = null; };
  }, [graphData]);

  // Continue ticking each frame until alpha decays
  useFrame(() => {
    const sim = simRef.current;
    if (sim && sim.alpha() > sim.alphaMin()) sim.tick();
  });

  return simNodesRef;
}
```

### Task 8: NebulaMesh — theme clouds with fresnel shader

Dynamic bounding sphere computed each frame from member project positions, with an additive fresnel glow shader. Additive blending makes overlapping clouds brighter at their intersection.

**Files:**
- Create: `src/components/Canvas/NebulaMesh.jsx`

- [ ] **Step 1: Create src/components/Canvas/NebulaMesh.jsx**

```jsx
import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';
import { useGraphStore } from '../../store/useGraphStore.js';

const vertexShader = /* glsl */`
  varying vec3 vNormal;
  varying vec3 vViewPosition;
  void main() {
    vNormal = normalize(normalMatrix * normal);
    vec4 mvPosition = modelViewMatrix * vec4(position, 1.0);
    vViewPosition = -mvPosition.xyz;
    gl_Position = projectionMatrix * mvPosition;
  }
`;

const fragmentShader = /* glsl */`
  uniform vec3 cloudColor;
  uniform float opacity;
  varying vec3 vNormal;
  varying vec3 vViewPosition;
  void main() {
    vec3 normal = normalize(vNormal);
    vec3 viewDir = normalize(vViewPosition);
    float fresnel = pow(1.0 - max(dot(normal, viewDir), 0.0), 3.0);
    gl_FragColor = vec4(cloudColor, fresnel * 0.8 * opacity);
  }
`;

const CLOUD_COLOR = new THREE.Color('#e8a030');
const _sphere = new THREE.Sphere();
const _box = new THREE.Box3();
const _v = new THREE.Vector3();

export function NebulaMesh({ themeNode, memberProjectIds, simNodesRef, graphNodes }) {
  const meshRef = useRef();
  const uniforms = useMemo(() => ({
    cloudColor: { value: CLOUD_COLOR },
    opacity: { value: 1.0 },
  }), []);

  // Pre-build index list of project nodes that are members of this theme
  const memberIndices = useMemo(() => {
    const indices = [];
    graphNodes.forEach((n, i) => {
      if (memberProjectIds.has(n.id)) indices.push(i);
    });
    return indices;
  }, [graphNodes, memberProjectIds]);

  useFrame(() => {
    if (!meshRef.current || memberIndices.length === 0) return;

    const simNodes = simNodesRef.current;

    // Compute bounding sphere from member project positions
    const points = memberIndices
      .map(i => simNodes[i])
      .filter(Boolean)
      .map(n => _v.set(n.x ?? 0, n.y ?? 0, n.z ?? 0).clone());

    if (points.length === 0) return;

    _box.setFromPoints(points);
    _box.getBoundingSphere(_sphere);

    // Add padding so the cloud visually envelops its nodes
    const radius = Math.max(_sphere.radius + 20, 30);
    meshRef.current.position.copy(_sphere.center);
    meshRef.current.scale.setScalar(radius);

    // Focus dimming
    const { focusNodeId, focusSet } = useGraphStore.getState();
    const inFocus = !focusNodeId || focusSet.has(themeNode.id);
    uniforms.opacity.value = inFocus ? 1.0 : 0.04;
  });

  return (
    <mesh ref={meshRef}>
      <sphereGeometry args={[1, 32, 24]} />
      <shaderMaterial
        vertexShader={vertexShader}
        fragmentShader={fragmentShader}
        uniforms={uniforms}
        transparent
        depthWrite={false}
        side={THREE.DoubleSide}
        blending={THREE.AdditiveBlending}
      />
    </mesh>
  );
}
```

### Task 9: ParticleSystem — oversees particles

Amber instanced points flowing from overseer toward target along each oversees link. Points reset position to source when they reach the target.

**Files:**
- Create: `src/components/Canvas/ParticleSystem.jsx`

- [ ] **Step 1: Create src/components/Canvas/ParticleSystem.jsx**

```jsx
import { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import * as THREE from 'three';

const PARTICLE_COUNT = 3;
const PARTICLE_SPEED = 0.004;

export function ParticleSystem({ link, simNodesRef, graphNodes }) {
  const pointsRef = useRef();

  // Pre-find source/target indices once
  const { srcIdx, tgtIdx } = useMemo(() => ({
    srcIdx: graphNodes.findIndex(n => n.id === (link.source?.id ?? link.source)),
    tgtIdx: graphNodes.findIndex(n => n.id === (link.target?.id ?? link.target)),
  }), [graphNodes, link]);

  // Each particle has an independent progress value (0 → 1)
  const progress = useMemo(
    () => Array.from({ length: PARTICLE_COUNT }, (_, i) => i / PARTICLE_COUNT),
    []
  );

  const positions = useMemo(
    () => new Float32Array(PARTICLE_COUNT * 3),
    []
  );

  useFrame(() => {
    const simNodes = simNodesRef.current;
    const src = simNodes[srcIdx];
    const tgt = simNodes[tgtIdx];
    if (!src || !tgt || !pointsRef.current) return;

    for (let i = 0; i < PARTICLE_COUNT; i++) {
      progress[i] = (progress[i] + PARTICLE_SPEED) % 1;
      const t = progress[i];
      positions[i * 3]     = src.x + (tgt.x - src.x) * t;
      positions[i * 3 + 1] = src.y + (tgt.y - src.y) * t;
      positions[i * 3 + 2] = src.z + (tgt.z - src.z) * t;
    }

    pointsRef.current.geometry.attributes.position.needsUpdate = true;
  });

  return (
    <points ref={pointsRef}>
      <bufferGeometry>
        <bufferAttribute
          attach="attributes-position"
          args={[positions, 3]}
        />
      </bufferGeometry>
      <pointsMaterial color="#f0a842" size={2} transparent opacity={0.7} depthWrite={false} />
    </points>
  );
}
```

```jsx
export function LinkLines({ links, simNodesRef, graphNodes, rawData }) {
  const linesRef = useRef();

  // Pre-compute index pairs and link metadata once
  const linkMeta = useMemo(() => links.map(link => {
    const srcId = link.source?.id ?? link.source;
    const tgtId = link.target?.id ?? link.target;
    return {
      srcIdx: graphNodes.findIndex(n => n.id === srcId),
      tgtIdx: graphNodes.findIndex(n => n.id === tgtId),
      color: getLinkColor(link, rawData),
    };
  }), [links, graphNodes, rawData]);

  const positions = useMemo(() => new Float32Array(links.length * 6), [links.length]);
  const colors = useMemo(() => {
    const arr = new Float32Array(links.length * 6);
    linkMeta.forEach(({ color }, i) => {
      const c = new THREE.Color(color);
      // Two vertices per line segment, each gets RGB
      for (let v = 0; v < 2; v++) {
        arr[(i * 2 + v) * 3]     = c.r;
        arr[(i * 2 + v) * 3 + 1] = c.g;
        arr[(i * 2 + v) * 3 + 2] = c.b;
      }
    });
    return arr;
  }, [linkMeta]);

  useFrame(() => {
    const simNodes = simNodesRef.current;
    if (!linesRef.current || !simNodes.length) return;

    const { focusNodeId, focusSet } = useGraphStore.getState();

    linkMeta.forEach(({ srcIdx, tgtIdx }, i) => {
      const src = simNodes[srcIdx];
      const tgt = simNodes[tgtIdx];
      if (!src || !tgt) return;

      positions[i * 6]     = src.x ?? 0;
      positions[i * 6 + 1] = src.y ?? 0;
      positions[i * 6 + 2] = src.z ?? 0;
      positions[i * 6 + 3] = tgt.x ?? 0;
      positions[i * 6 + 4] = tgt.y ?? 0;
      positions[i * 6 + 5] = tgt.z ?? 0;
    });

    linesRef.current.geometry.attributes.position.needsUpdate = true;
  });

  if (links.length === 0) return null;

  return (
    <lineSegments ref={linesRef}>
      <bufferGeometry>
        <bufferAttribute attach="attributes-position" args={[positions, 3]} />
        <bufferAttribute attach="attributes-color" args={[colors, 3]} />
      </bufferGeometry>
      <lineBasicMaterial vertexColors transparent opacity={0.7} />
    </lineSegments>
  );
}
```
